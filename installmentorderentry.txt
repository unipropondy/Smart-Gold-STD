    Private Sub frmInstallment_AJ_Load(sender As System.Object, e As System.EventArgs) Handles MyBase.Load
        Dim InsDate As Date
        If sMainFrom <> "MAINORDER" And sMainFrom <> "MAININS" Then
            PnlSalesManPwd.Visible = True
            If txtSalesManPWD.Enabled Then txtSalesManPWD.Focus()
            optInstallment.Checked = True
            getServerDate()
            getCurrentDate()
            optInstallment_CheckedChanged(Nothing, Nothing)
            subGetCompany()
            bEdit = False
            txtIDD.Text = ""
            txtIMM.Text = ""
            txtIYYYY.Text = ""
            InsDate = DateAdd("m", Val(2), Now())

            If IsDate(InsDate) Then
                txtIDD.Text = Format(InsDate, "dd")
                txtIMM.Text = Format(Month(InsDate), "00")
                txtIYYYY.Text = Format(Year(InsDate), "0000")
            End If
            ''grdInstallmentGrid.Rows.Clear()
        ElseIf sMainFrom = "MAINORDER" Then
            optOrder.Checked = True
            optInstallment.Checked = False
            txtInvoiceNo.Text = Trim(sMainOrderNo)
            txtInvoiceNo_LostFocus(sender, e)
        ElseIf sMainFrom = "MAININS" Then
            optInstallment.Checked = True
            optOrder.Checked = False
            txtInvoiceNo.Text = Trim(sMainOrderNo)
            txtInvoiceNo_LostFocus(sender, e)
        End If
        If optInstallment.Checked Then
            btnDelete.Text = "Item Transfer"
        End If

    Private Function fncCheckExisting(ByVal sinventorycode As String) As Boolean
        Dim icnt As Long
        Try
            fncCheckExisting = False
            For icnt = 0 To grdInstallmentGrid.RowCount - 1
                If Trim(fnGetdgvText(grdInstallmentGrid, icnt, 0)) = Trim(sinventorycode) Then
                    fncCheckExisting = True
                    Exit For
                End If
            Next icnt
        Catch ex As Exception
            MessageBox.Show(ex.Message)
        End Try
    End Function

    Private Function fncGetInventory(ByVal sinventorycode As String) As Boolean
        Dim lSalesBillNo&
        Dim dsInv As New DataSet
        Try
            dsInv = objIns.GetInventoryViewDetail(sinventorycode)
            fncGetInventory = True
            If dsInv.Tables(0).Rows.Count > 0 Then
                lSalesBillNo = Val(dsInv.Tables(0).Rows(0)("SalesBillNo") & "")
                If UCase(sCompanyShortCode = "GMT") Then
                    If (Val(dsInv.Tables(0).Rows(0)("QtyOnHand") & "") = 0) And (Trim(dsInv.Tables(0).Rows(0)("CategoryCode")) <> "22KASR") Then
                        MsgBox("Item Already Sold for Invoice " & Trim(dsInv.Tables(0).Rows(0)("SalesBillNo") & "") & " on " & Format(dsInv.Tables(0).Rows(0)("SalesDAte"), "dd/MM/yyyy"))
                        fncGetInventory = False
                        txtInventoryCode.Focus()
                        Exit Function
                    End If
                    If (Val(dsInv.Tables(0).Rows(0)("QtyOnHand") & "") < 1) And (Trim(dsInv.Tables(0).Rows(0)("CategoryCode")) <> "22KASR") Then
                        MsgBox("No Stock for this Inventory!")
                        fncGetInventory = False
                        txtInventoryCode.Focus()
                        Exit Function
                    End If
                Else
                    If ((Val(dsInv.Tables(0).Rows(0)("QtyOnHand") & "") = 0) And (Val(dsInv.Tables(0).Rows(0)("QtySold") & "") = 0)) Then
                        MsgBox("No Stock for this Inventory!")
                        fncGetInventory = False
                        txtInventoryCode.Focus()
                        Exit Function
                    End If

                    If (Val(dsInv.Tables(0).Rows(0)("QtyOnHand") & "") = 0) Then
                        MsgBox("Item Already Sold for Invoice " & Trim(dsInv.Tables(0).Rows(0)("SalesBillNo") & "") & " on " & Format(dsInv.Tables(0).Rows(0)("SalesDAte"), "dd/MM/yyyy"))
                        fncGetInventory = False
                        txtInventoryCode.Focus()
                        Exit Function
                    End If
                    If (Val(dsInv.Tables(0).Rows(0)("QtyOnHand") & "") < 1) Then
                        MsgBox("No Stock for this Inventory!")
                        fncGetInventory = False
                        txtInventoryCode.Focus()
                        Exit Function
                    End If

                End If

                If Trim$(dsInv.Tables(0).Rows(0)("LocationCode") & "") <> gthisLocation Then
                    MsgBox("The selected Inventory belongs to the Location - " & Trim$(dsInv.Tables(0).Rows(0)("LocationCode") & "") & "!")
                    fncGetInventory = False
                    txtInventoryCode.Focus()
                    Exit Function
                End If
                If Trim$(dsInv.Tables(0).Rows(0)("Status") & "") = "IN" Then
                    MsgBox("This Inventory is in Installment Status!")
                    fncGetInventory = False
                    txtInventoryCode.Focus()
                    Exit Function
                End If
                If Trim$(dsInv.Tables(0).Rows(0)("Status") & "") = "OR" Then
                    MsgBox("This Inventory is in Order Status!")
                    fncGetInventory = False
                    txtInventoryCode.Focus()
                    Exit Function
                End If

                If Trim$(dsInv.Tables(0).Rows(0)("OrderNo") & "") <> "" Then
                    Dim dsOrder As New DataSet
                    Dim objOrder As New clsOrder
                    dsOrder = objOrder.GetOrderHeader(Trim$(dsInv.Tables(0).Rows(0)("OrderNo") & ""))
                    If dsOrder.Tables(0).Rows.Count > 0 Then
                        If txtCustomerCode.Text.Trim <> Trim(dsOrder.Tables(0).Rows(0)("CustomerCode")) Then
                            MessageBox.Show("This inventory is ordered for some other customer. you Cant sale this now.")
                            fncGetInventory = False
                            txtInventoryCode.Focus()
                            Exit Function
                        Else
                            dAdvancePaid = Val(dsOrder.Tables(0).Rows(0)("AdvancePaid"))
                            sOrderRefNo = Trim$(dsInv.Tables(0).Rows(0)("OrderNo") & "")
                            sClassCode = Trim(dsInv.Tables(0).Rows(0)("ClassCode") & "")
                            subGetClass()
                            If dsOrder.Tables(0).Rows(0)("RateFixed") = "Y" Then
                                txtCostPerGram.Text = Format(Val(dsOrder.Tables(0).Rows(0)("CostPerGram")), "0.00")
                            End If
                            txtInventoryDescription.Text = Trim(dsInv.Tables(0).Rows(0)("Description") & "")
                            If msClsSalesType = "P" Or msClsSalesType = "B" Then
                                txtPCS.Enabled = True
                                If UCase(sCompanyShortCode) = "ES" Or UCase(sCompanyShortCode) = "AS" Or UCase(sCompanyShortCode) = "HN" Then
                                    txtPCS.Text = 1
                                Else
                                    txtPCS.Text = Val(dsInv.Tables(0).Rows(0)("QtyOnHand") & "")
                                End If
                                If Val(dsInv.Tables(0).Rows(0)("Pcs")) > 0 Then
                                    txtWeight.Text = Format(Val(dsInv.Tables(0).Rows(0)("Weight") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & ""), "0.00")
                                    txtGoldLoss.Text = Format(Val(dsInv.Tables(0).Rows(0)("GoldLoss") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & ""), "0.00")
                                    txtSMC.Text = Format(Val(dsInv.Tables(0).Rows(0)("sMc") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & ""), "0.00")
                                    mdPurMC = Val(dsInv.Tables(0).Rows(0)("PMC") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & "")
                                    dTotalCost = Val(dsInv.Tables(0).Rows(0)("TotalCost") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & "")
                                Else
                                    txtWeight.Text = 0.0
                                    txtGoldLoss.Text = 0.0
                                    txtSMC.Text = 0.0
                                    mdPurMC = 0.0
                                    dTotalCost = 0.0
                                End If
                            Else
                                ''txtStoneWeight.Text = Format(Val(dsInv.Tables(0).Rows(0)("StoneWeight") & ""), "0.00")
                                txtPCS.Text = Val(dsInv.Tables(0).Rows(0)("Pcs") & "")
                                txtWeight.Text = Format(Val(dsInv.Tables(0).Rows(0)("Weight") & ""), "0.00")
                                txtGoldLoss.Text = Format(Val(dsInv.Tables(0).Rows(0)("GoldLoss") & ""), "0.00")
                                txtSMC.Text = Format(Val(dsInv.Tables(0).Rows(0)("sMc") & ""), "0.00")
                                mdPurMC = Val(dsInv.Tables(0).Rows(0)("PMC") & "")
                                dTotalCost = Val(dsInv.Tables(0).Rows(0)("TotalCost") & "")
                            End If
                        End If


                    End If
                Else
                    sClassCode = Trim(dsInv.Tables(0).Rows(0)("ClassCode") & "")
                    'load class details to get the rate and the salestype.
                    subGetClass()
                    txtInventoryDescription.Text = Trim(dsInv.Tables(0).Rows(0)("Description") & "")
                    If msClsSalesType = "P" Or msClsSalesType = "B" Then
                        txtPCS.Enabled = True
                        '' txtStoneWeight.Text = Format(Val(dsInv.Tables(0).Rows(0)("StoneWeight") & ""), "0.00")
                        ''HAJ AS MUKKUTHI Sales by PCS on 29 Aug 2007
                        If UCase(sCompanyShortCode) = "ES" Or UCase(sCompanyShortCode) = "AS" Or UCase(sCompanyShortCode) = "HN" Then
                            txtPCS.Text = 1
                            'ElseIf UCase(sCompanyShortCode) = "HN" Then
                            '    If txtPCS.Enabled = True Then txtPCS.Focus()
                        Else
                            txtPCS.Text = Val(dsInv.Tables(0).Rows(0)("QtyOnHand") & "")
                        End If
                        If Val(dsInv.Tables(0).Rows(0)("Pcs")) > 0 Then
                            txtWeight.Text = Format(Val(dsInv.Tables(0).Rows(0)("Weight") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & ""), "0.00")
                            txtGoldLoss.Text = Format(Val(dsInv.Tables(0).Rows(0)("GoldLoss") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & ""), "0.00")
                            txtSMC.Text = Format(Val(dsInv.Tables(0).Rows(0)("sMc") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & ""), "0.00")
                            mdPurMC = Val(dsInv.Tables(0).Rows(0)("PMC") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & "")
                            dTotalCost = Val(dsInv.Tables(0).Rows(0)("TotalCost") & "") * Val(txtPCS.Text) / Val(dsInv.Tables(0).Rows(0)("Pcs") & "")
                        Else
                            txtWeight.Text = 0.0
                            txtGoldLoss.Text = 0.0
                            txtSMC.Text = 0.0
                            mdPurMC = 0.0
                            dTotalCost = 0.0
                        End If
                    Else
                        ''txtStoneWeight.Text = Format(Val(dsInv.Tables(0).Rows(0)("StoneWeight") & ""), "0.00")
                        txtPCS.Text = Val(dsInv.Tables(0).Rows(0)("Pcs") & "")

        Catch ex As Exception
            MessageBox.Show(ex.Message)
        End Try
    End Sub

    Private Function fncPaymodeDetailSave(ByVal sOrderNo As String) As Boolean
        Dim objPM As New clsPayMode
        Try
            Dim bPayModeFlag As Boolean
            Dim sDate As String
            Dim i As Long
            Dim sSalesType As String
            bPayModeFlag = False
            sDate = Format(Now, "dd/MM/yyyy")
            If optInstallment.Checked Then
                sSalesType = "INSTALLMENT"
            Else
                sSalesType = "ORDER"
            End If
            For i = 0 To UBound(arrpay, 1)
                If arrpay(i, 1) > 0 Then
                    iRet = objPM.PayModeDetailSave(sOrderNo, sDate, Trim(arrpay(i, 3)), _
                                                 Val(arrpay(i, 1)), Val(arrpay(i, 2)), _
                                                 Trim(arrpay(i, 4)), Trim(sSalesType))
                    bPayModeFlag = True
                    If Not iRet Then
                        fncPaymodeDetailSave = False
                        Exit Function
                    End If

                End If
            Next i
            If bPayModeFlag = False Then
                iRet = objPM.PayModeDetailSave(sOrderNo, sDate, "CASH", _
                                            Val(txtDepositAmount.Text), 0, _
                                            "MANUAL", Trim(sSalesType))
                If Not iRet Then
                    fncPaymodeDetailSave = False
                    Exit Function
                End If
            End If
            fncPaymodeDetailSave = True
        Catch ex As Exception
            MessageBox.Show(ex.Message)
        End Try
    End Function

    Public Sub DispSupplierDetails(ByVal suplCode As String)
        Dim ds As DataSet
        Try
            If suplCode.Length > 0 Then
                ds = objClsSupplier.GetSupplierDetail(suplCode)
                If ds.Tables(0).Rows.Count > 0 Then
                    txtSupplierCode.Text = Trim(ds.Tables(0).Rows(0)("suppliercode") & "")
                    txtSupplierName.Text = Trim(ds.Tables(0).Rows(0)("SupplierName") & "")
                    txtSAddress1.Text = Trim(ds.Tables(0).Rows(0)("Address1") & "")
                    txtSAddress2.Text = Trim(ds.Tables(0).Rows(0)("Address2") & "")
                    txtSAddress3.Text = Trim(ds.Tables(0).Rows(0)("Address3") & "")
                    txtSupplierName.ReadOnly = True
                    txtSAddress1.ReadOnly = True
                    txtSAddress2.ReadOnly = True
                    txtSAddress3.ReadOnly = True
                    'txtClass.Focus()
                Else
                    MsgBox("No such Supplier code ! ", MsgBoxStyle.Exclamation)
                    txtSupplierCode.Clear()
                    txtSupplierCode.Focus()
                    Exit Sub
                End If
            End If
        Catch ex As Exception
            MsgBox(Err.Description)
        Finally
            Err.Clear()
            ds = Nothing
        End Try
    End Sub

                    MessageBox.Show("Item Transferred.")
                    ClearControls()
                Else
                    sOrderStatus = objOrder.GetOrderStatus(txtInvoiceNo.Text.Trim)
                    If sOrderStatus.Trim = "" Then
                        MessageBox.Show("Order number not exists.")
                    ElseIf UCase(sOrderStatus.Trim) = PURCHASED Then
                        MessageBox.Show("This ordered items are already purchased. You cant delete this order.")
                    ElseIf UCase(sOrderStatus.Trim) = ORDERED Or UCase(sOrderStatus.Trim) = REPAIR Then
                        objOrder.OpenOrderConnection()
                        objOrder.gpBeginOrderTransaction()
                        If Not objOrder.OrderDelete(txtInvoiceNo.Text.Trim) Then
                            MessageBox.Show("Error in Delete.")
                            objOrder.gpRollBackOrderConnection()
                            objOrder.CloseOrderConnection()
                        End If
                        'update payment detail
                        dAmount = objOrder.GetOrderPaymentAmount(txtInvoiceNo.Text.Trim)
                        If Not objPM.PayModeDetailSave(txtInvoiceNo.Text.Trim, Format(Now, "dd/MM/yyyy"), "CASH", -dAmount, 0, "", "ORDER") Then
                            MessageBox.Show("Error in Delete.")
                            objOrder.gpRollBackOrderConnection()
                            objOrder.CloseOrderConnection()
                        End If
                        'delete from table orderpayment
                        If Not objOrder.fncOrderPaymentDetailDelete(txtInvoiceNo.Text.Trim, "", 0) Then
                            MessageBox.Show("Error in Delete.")
                            objOrder.gpRollBackOrderConnection()
                            objOrder.CloseOrderConnection()
                        End If
                        objOrder.gpCommitOrderTransaction()
                        objOrder.CloseOrderConnection()
                        MessageBox.Show("Order Deleted.")
                    End If
                    ClearControls()

                End If
            End If
        Catch ex As Exception
            MessageBox.Show(ex.Message)
        Finally
            objOrder.CloseOrderConnection()
            objOrder = Nothing

                    nTotalAmt = nTotalAmt + Val(fnGetdgvText(grdSales, iCnt, ITotalValue))

                          Val(txtCostPerGram.Text), Val(nWeight), _
                          Val(nStoneWeight), Val(nSMC), Val(nPCS), _
                          Val(nGoldLoss), Val(nTotCost), _
                          -Val(nTotalAmt), "TRADEIN", "N", sUserID)
                If Not bStatus Then
                    Exit Function
                End If
                i = 0
                For iCnt = 0 To grdSales.RowCount - 1
                    If fnGetdgvText(grdSales, iCnt, ISalesType).Trim.ToUpper = "TI" Then
                        sTIinventorycode = "TI-" & sInvoiceNo & "-" & i + 1

                        s916 = Format(Val(dsClass.Tables(0).Rows(i)("Rate") & ""), "#0.00")
