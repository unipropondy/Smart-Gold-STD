    Private Sub frmTransferReports_KeyDown(sender As Object, e As System.Windows.Forms.KeyEventArgs) Handles Me.KeyDown

            Case Keys.F7
                btnOK_Click(sender, e)
            Case Keys.Escape
                btnClose_Click(sender, e)
            Case Keys.Return
                SendKeys.Send("{Tab}")
            Case Keys.Down
                SendKeys.Send("{Tab}")
            Case Keys.Up
                SendKeys.Send("+{Tab}")

    Public Sub AddItemstoComboFromLoc()
        'dm.SetMasterDataSetLoc("Select rtrim(LocationCode)LocationCode,rtrim(LocationName)LocationName from tblLocation Order by SortCode", "tbllocation")
        'dm.FillDatasToComboListWithFilters(cboFromLocationCode, "tbllocation", "LocationCode", "LocationName")

        dm.SetMasterDataSetLoc("Select 'ALL' LocationCode,'ALL' LocationName Union Select LocationCode,LocationName from tblLocation", "tbllocation")
        dm.FillDatasToComboListWithFilters(cboFromLocationCode, "tbllocation", "LocationCode", "LocationName")
    End Sub
    Public Sub AddItemstoComboToLoc()
        'dm.SetMasterDataSetLoc("Select rtrim(LocationCode)LocationCode,rtrim(LocationName)LocationName from tblLocation Order by SortCode", "tbllocation")
        'dm.FillDatasToComboListWithFilters(cboToLocationCode, "tbllocation", "LocationCode", "LocationName")

        dm.SetMasterDataSetLoc("Select 'ALL' LocationCode,'ALL' LocationName Union Select LocationCode,LocationName from tblLocation", "tbllocation")
        dm.FillDatasToComboListWithFilters(cboToLocationCode, "tbllocation", "LocationCode", "LocationName")
    End Sub
    Public Sub LoadLocationcode()
        If gthisLocationType = "HQ" Then
            If cboFromLocationCode.Items.Count > 0 Then cboFromLocationCode.SelectedIndex = -1
            cboFromLocationCode.Text = gthisLocation
            If cboToLocationCode.Items.Count > 0 Then cboToLocationCode.SelectedIndex = -1
            cboToLocationCode.Text = gthisLocation
        Else
            'If cboFromLocationCode.Items.Count > 0 Then cboFromLocationCode.Text = gthisLocation
            'If cboToLocationCode.Items.Count > 0 Then cboToLocationCode.Text = gthisLocation
            If cboFromLocationCode.Items.Count > 0 Then cboFromLocationCode.SelectedIndex = -1
            cboFromLocationCode.Text = gthisLocation
            If cboToLocationCode.Items.Count > 0 Then cboToLocationCode.SelectedIndex = -1
            cboToLocationCode.Text = gthisLocation
        End If
    End Sub
