    Private Function fncValidate() As Boolean
        If UCase(txtCustomerCode.Text) = "CASH" And sMemberModule = "1" Then
            If txtMemberCode.Text.Trim = "" Then
                MsgBox("MemberCode is NOT entered !", MsgBoxStyle.Critical)
                txtCustomerCode.Focus()
                fncValidate = False
                Exit Function
            End If
        ElseIf UCase(txtCustomerCode.Text) = "CASH" And sMemberModule = "0" Then
            MessageBox.Show(" TradeIn Sales Not for CASH Customer!!!")
            txtCustomerCode.Focus()
            fncValidate = False
            Exit Function
        ElseIf Trim(txtCustomerCode.Text.Trim) = "" Then
            MessageBox.Show(" Choose Customer!!!")
            txtCustomerCode.Focus()
            fncValidate = False
            Exit Function
        End If
        If grdTrade.RowCount < 0 Then
            MsgBox("Enter trade in item details!!", vbCritical)
            fncValidate = False
            Exit Function
        End If

        If Val(lblTradeInAmount.Text) <= 0 Then
            MsgBox("TradeIn Amount has some Value!!!", vbCritical)
            fncValidate = False
            Exit Function
        End If
        fncValidate = True
    End Function
    Private Function fncSaveTradeIn() As Boolean
        Dim SBll As New clsTradeIN
        Dim SBlm As New clsInventory
        Dim frmPm As New clsPayMode
        Dim adoRS As ADODB.Recordset
        Dim sError As String
        Dim iRet As Boolean

        Dim i As Long
        Dim sSql As String
        If fncGetInvoiceNo() = False Then
            MessageBox.Show("Transaction failed!!!")
            Exit Function
        End If
        iRet = SBll.TradeInDelete(oTSalesConn, oTSalesRs, Trim(sinventorycode), sError)
        If Trim(sError) = "" Then
            dPaidAmount = Val(fnGetdgvText(grdTrade, grdTrade.RowCount - 1, 7)) + (Val(fnGetdgvText(grdTrade, grdTrade.RowCount - 1, 7)) * 7) / 100
            iRet = SBll.TradeInHeaderSave(oTSalesConn, oTSalesRs, Trim(sinventorycode), Trim$(sInvoiceNo), _
                      gthisLocation, UCase(Trim(txtCustomerCode.Text)), Trim(txtMemberCode.Text), Trim(txtCustomerName.Text), Trim(txtAddress1.Text), _
                      Trim(txtAddress2.Text), Trim(txtAddress3.Text), _
                      Val(txtCostPerGram.Text), Val(fnGetdgvText(grdTrade, grdTrade.RowCount - 1, 2)), _
                      Val(fnGetdgvText(grdTrade, grdTrade.RowCount - 1, 9)), _
                      Val(fnGetdgvText(grdTrade, grdTrade.RowCount - 1, 5)), Val(fnGetdgvText(grdTrade, grdTrade.RowCount - 1, 6)), _
                      Val(fnGetdgvText(grdTrade, grdTrade.RowCount - 1, 3)), Val(fnGetdgvText(grdTrade, grdTrade.RowCount - 1, 7)), _
                      Val(dPaidAmount), "TRADEIN", "N", msSalesManCode, sError)
            If Trim(sError) <> "" Then
                MsgBox(sError, vbCritical)
                fncSaveTradeIn = False
                Exit Function
            End If

            For i = 0 To grdTrade.RowCount - 1
                If Not Trim(fnGetdgvText(grdTrade, i, 1)) = "TOTAL:" Then
                    iRet = SBll.TradeInDetailSave(oTSalesConn, oTSalesRs, gthisLocation, Trim(sinventorycode), i, Trim(fnGetdgvText(grdTrade, i, 1)), _
                            Trim(fnGetdgvText(grdTrade, i, 8)), Val(fnGetdgvText(grdTrade, i, 2)), Val(fnGetdgvText(grdTrade, i, 9)), _

                End If
                If Trim(sError) <> "" Then
                    MsgBox(sError, vbCritical)
                    fncSaveTradeIn = False
                    Exit Function
                End If
            Next
            If Not fncPaymodeDetailSave(sInvoiceNo) Then
                MessageBox.Show("Transaction not saved")
                fncSaveTradeIn = False
                Exit Function
            End If

            If Trim(sError) <> "" Then
                MsgBox(sError, vbCritical)
                fncSaveTradeIn = False
                Exit Function
            Else
                fncSaveTradeIn = True
                Exit Function
            End If
        Else
            MsgBox(sError, vbCritical)
            fncSaveTradeIn = False
            Exit Function
        End If
        frmPm = Nothing

    End Function
    Private Sub subGetTradeInDetails()
        Dim i As Long
        Dim adoRS As DataSet
        Dim SBll As New clsTradeIN

        adoRS = SBll.GetTradeInList(sinventorycode)

        grdTrade.RowCount = 0

        On Error GoTo ErrorSub

    Private Function fncValidateTradeIn() As Boolean
        If Trim(txtDescription.Text) = "" Then
            MsgBox("Description Cannot be Blank ....")
            txtDescription.Focus()
            fncValidateTradeIn = False
            Exit Function
        End If

        If IsNumeric(txtWeight.Text) = False Then
            MsgBox("Weight is Invalid ....")
            fncValidateTradeIn = False
            txtWeight.Focus()
            Exit Function
        End If

        If Val(txtWeight.Text) <= 0 Then
            MsgBox("Weight is Invalid ....")
            fncValidateTradeIn = False
            txtWeight.Focus()
            Exit Function
        End If
        If Trim(txtStoneWeight.Text) <> "" Then
            If IsNumeric(txtStoneWeight.Text) = False Then
                MsgBox("Stone Weight is Invalid ....")
                fncValidateTradeIn = False
                txtStoneWeight.Focus()
                Exit Function
            End If

            If Val(txtStoneWeight.Text) < 0 Then
                MsgBox("Stone Weight is Invalid ....")
                fncValidateTradeIn = False
                txtStoneWeight.Focus()
                Exit Function
            End If
        End If
        If Val(txtGoldLoss.Text) < 0 Then
            MsgBox("Gold Loss is Invalid ....")
            fncValidateTradeIn = False
            txtGoldLoss.Focus()
            Exit Function
        End If

        If IsNumeric(txtCostPerGram1.Text) = False Then
            MsgBox("Cost/Gram is Invalid ....")
            fncValidateTradeIn = False
            txtCostPerGram1.Focus()
            Exit Function
        End If

        If Val(txtCostPerGram1.Text) <= 0 Then
            MsgBox("Cost Per Gram is Invalid ....")
            fncValidateTradeIn = False
            txtCostPerGram1.Focus()
            Exit Function
        End If
        fncValidateTradeIn = True

        Exit Function

    End Function

    Private Sub subGetCostPerGram()

        On Error GoTo ErrHandler

        End If
        ssql = "Select Rate from tblClass where ClassCode like '22%'"
        'GetRecordset(gsConn, ssql, adoRS)
        adoRS = dm.getDataSetWithSqlAdapter(ssql, "Table", "Table")
        If adoRS.Tables(0).Rows.Count = 0 Then
            ssql = "Select Rate from tblClass where ClassCode like '18K%'"
            'GetRecordset(gsConn, ssql, adoRS)
            adoRS = dm.getDataSetWithSqlAdapter(ssql, "Table", "Table")
            If adoRS.Tables(0).Rows.Count = 0 Then
                ssql = "Select Rate from tblClass"
                'GetRecordset(gsConn, ssql, adoRS)
                adoRS = dm.getDataSetWithSqlAdapter(ssql, "Table", "Table")
                If adoRS.Tables(0).Rows.Count > 0 Then
                    txtCostPerGram.Text = Val(adoRS.Tables(0).Rows(0)("Rate") & "")
                    txtCostPerGram1.Text = Val(txtCostPerGram.Text) - (Val(txtCostPerGram.Text) * Val(txtLossPerc.Text) / 100)

        Else
            txtCostPerGram.Text = Val(adoRS.Tables(0).Rows(0)("Rate") & "")
            txtCostPerGram1.Text = Val(txtCostPerGram.Text) - (Val(txtCostPerGram.Text) * Val(txtLossPerc.Text) / 100)
        End If
        Exit Sub
ErrHandler:
        MsgBox(Err.Description)
        Err.Clear()
    End Sub
